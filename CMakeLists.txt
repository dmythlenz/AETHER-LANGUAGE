cmake_minimum_required(VERSION 3.16)
project(aetherc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(aether STATIC src/aether.cpp)
target_include_directories(aether PUBLIC include)
add_executable(aetherc src/main.cpp)
target_link_libraries(aetherc PRIVATE aether)
add_executable(aether_smoke tests/smoke.cpp)
target_link_libraries(aether_smoke PRIVATE aether)
add_executable(aether_benchmark tests/benchmark.cpp)
target_link_libraries(aether_benchmark PRIVATE aether)
enable_testing()
add_test(NAME aether_smoke COMMAND aether_smoke)
